// ===================== UTIL =====================
function rnd(a,b){return a+Math.random()*(b-a)}
function pick(a){return a[Math.floor(Math.random()*a.length)]}
function dist3(a,b){return Math.hypot(a[0]-b[0],a[1]-b[1],a[2]-b[2])}
function dirYP(y,p){return [-Math.sin(y)*Math.cos(p),Math.sin(p),-Math.cos(y)*Math.cos(p)]}
function norm(v){const l=Math.hypot(v[0],v[1],v[2])||1;return [v[0]/l,v[1]/l,v[2]/l]}
function rayBox(o,d,mn,mx,maxT){let t0=0,t1=maxT;for(let i=0;i<3;i++){if(Math.abs(d[i])<1e-9){if(o[i]<mn[i]||o[i]>mx[i])return -1}else{let a=(mn[i]-o[i])/d[i],c=(mx[i]-o[i])/d[i];if(a>c){const t=a;a=c;c=t}if(a>t0)t0=a;if(c<t1)t1=c;if(t0>t1)return -1}}return t0}
// v7: round columns are hit as what you see - a shaft cylinder (r*.92) between square base/capital slabs, not the full square box
function colRay(o,d,b,maxT){const sx=b.max[0]-b.min[0],sz=b.max[2]-b.min[2];const cx=(b.min[0]+b.max[0])/2,cz=(b.min[2]+b.max[2])/2,R=Math.min(sx,sz)/2*.92,y0=b.min[1]+.18,y1=b.max[1]-.18;let best=-1;const take=t=>{if(t>=0&&t<maxT&&(best<0||t<best))best=t};
 take(rayBox(o,d,b.min,[b.max[0],y0,b.max[2]],maxT));take(rayBox(o,d,[b.min[0],y1,b.min[2]],b.max,maxT));
 const ox=o[0]-cx,oz=o[2]-cz,a=d[0]*d[0]+d[2]*d[2];if(a>1e-9){const bb=ox*d[0]+oz*d[2],c=ox*ox+oz*oz-R*R,disc=bb*bb-a*c;if(disc>=0){let t=(-bb-Math.sqrt(disc))/a;if(t<0&&c<=0)t=0;if(t>=0){const y=o[1]+d[1]*t;if(y>=y0&&y<=y1)take(t)}}}else if(ox*ox+oz*oz<=R*R){const t=d[1]>0?(y0-o[1])/d[1]:(y1-o[1])/d[1];if(t>=0)take(t)}return best}
function isColBox(b){return b.tex==='column'&&Math.abs((b.max[0]-b.min[0])-(b.max[2]-b.min[2]))<.05}
function rayWorld(o,d,maxT){let best=maxT;for(const b of G.boxes){if(b.liquid)continue;let t=rayBox(o,d,b.min,b.max,best);if(t>=0&&b.colR===undefined)b.colR=isColBox(b);if(t>=0&&b.colR)t=colRay(o,d,b,best);if(t>=0&&t<best)best=t}if(d[1]<0){const t=-o[1]/d[1];if(t<best)best=t}return best}
function segSmoke(a,b){const d=[b[0]-a[0],b[1]-a[1],b[2]-a[2]],L=Math.hypot(...d);if(L<1e-6)return false;for(const s of G.smokes){if(s.r<.2)continue;const R=s.r*1.25;/*v8: matches the drawn cloud (sprites reach ~1.3-1.6r), was .89r so bots saw through the visible edge*/const t=Math.max(0,Math.min(L,((s.pos[0]-a[0])*d[0]+(s.pos[1]-a[1])*d[1]+(s.pos[2]-a[2])*d[2])/L));const px=a[0]+d[0]/L*t-s.pos[0],py=a[1]+d[1]/L*t-s.pos[1],pz=a[2]+d[2]/L*t-s.pos[2];if(px*px+py*py+pz*pz<R*R)return true}return false}
function los(a,b,smoke){const d=[b[0]-a[0],b[1]-a[1],b[2]-a[2]],L=Math.hypot(...d);const t=rayWorld(a,norm(d),L);if(t<L-.05)return false;if(smoke&&segSmoke(a,b))return false;return true}
function genCode(){const c='ABCDEFGHJKLMNPQRSTUVWXYZ';let s='';for(let i=0;i<5;i++)s+=c[Math.floor(Math.random()*c.length)];return s}
function pH(p){return p.crouch?1.2:1.8}
function eye(p){return [p.pos[0],p.pos[1]+pH(p)-.15,p.pos[2]]}
function chest(p){return [p.pos[0],p.pos[1]+pH(p)*.65,p.pos[2]]}
// per-part hitboxes in the player's local frame (model faces -z); must match buildPlayerMesh proportions
function hipY(p){return p.crouch?.42:.85}
function partBoxes(q){const hy=hipY(q);return [['head',[-.15,hy+.6,-.15],[.15,hy+.95,.15]],['body',[-.24,hy,-.15],[.24,hy+.6,.15]],['limb',[.24,hy+.1,-.16],[.38,hy+.6,.16]],['limb',[-.38,hy+.1,-.16],[-.24,hy+.6,.16]],['limb',[-.22,0,-.14],[.22,hy,.14]]]}
function hitPlayer(q,o,d,maxT){const c=Math.cos(q.yaw),s=Math.sin(q.yaw);const rx=o[0]-q.pos[0],ry=o[1]-q.pos[1],rz=o[2]-q.pos[2];
 const lo=[rx*c-rz*s,ry,rx*s+rz*c],ld=[d[0]*c-d[2]*s,d[1],d[0]*s+d[2]*c];let best=maxT,part=null;
 for(const [n,mn,mx] of partBoxes(q)){const t=rayBox(lo,ld,mn,mx,best);if(t>=0&&t<best){best=t;part=n}}return part?{t:best,part}:null}
function hitScan(o,d,maxT,excl){let best=rayWorld(o,d,maxT),who=-1,part=null;for(const q of G.P){if(q.id===excl||!q.alive)continue;const h=hitPlayer(q,o,d,best);if(h){best=h.t;who=q.id;part=h.part}}return {t:best,who,part}}
const PART_I={head:0,body:1,limb:2},PARTS=['head','body','limb'];
function wrapA(a){while(a>Math.PI)a-=2*Math.PI;while(a<-Math.PI)a+=2*Math.PI;return a}
