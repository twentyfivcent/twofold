function inSolid(q){if(q[1]<=.02)return true;for(const b of G.boxes){if(b.liquid)continue;if(q[0]>b.min[0]&&q[0]<b.max[0]&&q[1]>b.min[1]&&q[1]<b.max[1]&&q[2]>b.min[2]&&q[2]<b.max[2])return true}return false}
function penTrace(o,d,t0,pen,pid){let t=t0+.01;const lim=t0+pen+.02;while(t<lim){const q=[o[0]+d[0]*t,o[1]+d[1]*t,o[2]+d[2]*t];if(!inSolid(q))break;t+=.03}if(t>=lim)return null;const th=Math.max(.01,t-t0);if(o[1]+d[1]*t<=.02)return null;
 const o2=[o[0]+d[0]*(t+.01),o[1]+d[1]*(t+.01),o[2]+d[2]*(t+.01)];const r2=hitScan(o2,d,200-t,pid);if(r2.who<0)return null;r2.t+=t+.01;return {r:r2,th,mult:Math.max(.35,.75-.4*th/pen)}}

// v6 flag spawns: 3 validated points per map (0 = centre). Host picks one per round at random, never the same twice in a row.

// v6 MVP: host-computed top match score across humans AND bots. Score = kills*100 + damage*.5 + assists*40 + flag caps*150 + flag-hold seconds*10.
function mvpScore(p){return Math.round((p.k|0)*100+(p.dmg||0)*.5+(p.as|0)*40+(p.fc|0)*150+(p.fp||0)*10)}
function computeMVP(){let best=null;for(const p of G.P){if(p.off)continue;const sc=mvpScore(p);if(!best||sc>best.score)best={id:p.id,score:sc,bot:!!p.bot,name:p.name}}return best||{id:-1,score:0,bot:true}}
