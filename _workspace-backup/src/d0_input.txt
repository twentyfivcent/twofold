// v8 mouse spike filter: Chromium pointer-lock occasionally reports a single huge movementX/Y (hundreds of px) that the
// player never made -> a hard 'flick' while tracking. Drop a sample that is far above the recent average (real fast flicks build up over several events).

// v6 gun wall-tuck: rays from the eye along the aim (centre + gun side). Returns 0..1 (1 = muzzle fully inside wall reach).
const GUN_REACH=.85;
function gunTuck(p){if(!p||!p.alive||!G||!G.boxes)return 0;const e=[p.pos[0],p.pos[1]+pH(p)-.15,p.pos[2]],cy=Math.cos(p.yaw),sy=Math.sin(p.yaw),cp=Math.cos(p.pitch),sp=Math.sin(p.pitch);
 const d=[-sy*cp,sp,-cy*cp];let t=rayWorld(e,d,GUN_REACH);const o=[e[0]+cy*.17,e[1]-.12,e[2]-sy*.17];t=Math.min(t,rayWorld(o,d,GUN_REACH));
 const v=Math.max(0,Math.min(1,(GUN_REACH-t)/(GUN_REACH-.3)));p.tuck=v;return v}
