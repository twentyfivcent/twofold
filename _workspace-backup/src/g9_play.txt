// ---- bot takeover: dead human takes control of a living teammate bot for the rest of the round (host-authoritative, once per round)
function canTakeover(from,to){const a=G.P[from],b=G.P[to];return G.rs==='live'&&a&&b&&!a.alive&&b.alive&&b.bot&&!b.pendConn&&a.team===b.team&&!a.bot&&!(G.took&&G.took[a.tkId||from])}
function requestTakeover(to){if(HOST)hostTakeover(ME,to);else if(hostConn)netSend(hostConn,{t:'take',i:to})}
function hostTakeover(from,to){if(!canTakeover(from,to)){if(from===ME)addKF('<span style="color:#f99">Takeover unavailable</span>');return false}
 const a=G.P[from],b=G.P[to];G.took=G.took||{};const tk=a.tkId||from;G.took[tk]=1;
 const nm=a.name;b.bname=b.name;b.name=nm;a.name=nm+"'s bot";b.bot=false;b.bot_=null;a.bot=true;a.bot_=null;b.tkId=tk;a.tkId=undefined;b.tkFrom=from;b.tk0={k:b.k|0,d:b.d|0,dmg:b.dmg||0,as:b.as|0,fc:b.fc|0,fp:b.fp||0};
 b.conn=a.conn;a.conn=null;if(b.conn){b.conn.slot=to;netSend(b.conn,{t:'you',me:to})}
 if(from===ME){ME=to;onBecome()}
 emit({t:'n6',h:'<span style="color:#7fd0ff">'+esc(nm)+' took control of a teammate bot</span>'});return true}
